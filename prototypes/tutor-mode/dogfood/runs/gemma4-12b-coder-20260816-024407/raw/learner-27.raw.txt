The anvil: a sequence that is legally a `factor` but is semantically garbage on the boundary of a `term` — like `(5+2) <binary op here instead of factor>` when the code expects only what a primary/factor may genuinely produce. The check flags the mismatch exactly where the grammar rule's topology disagrees with the token stream.

Removing them now: the tests are green and the grammar is fully recursive, so the validation survives. The boundary guards (`evaluate`'s EOF and `primary`'s RPAREN) enforce the overall sentence, so the fat whitelists are dead weight and can be purged without changing behavior.

from collections import namedtuple

Token = namedtuple('Token', ['type', 'value'])

class ExpressionError(Exception):
    pass


def evaluate(expr):
    if not expr.strip():
        raise ExpressionError("Empty expression")

    tokens = lex(expr)

    if not tokens:
        raise ExpressionError("Could not parse")

    state = {'position': 0}

    def peek():
        return tokens[state['position']]

    def consume():
        token = peek()
        state['position'] += 1
        return token

    def expression():
        left = term()
        while state['position'] < len(tokens) and peek().type in ('PLUS', 'MINUS'):
            op = consume()
            right = term()
            left = left + right if op.type == 'PLUS' else left - right

        return left

    def term():
        left = factor()
        while state['position'] < len(tokens) and peek().type in ('STAR', 'SLASH'):
            op = consume()
            right = factor()
            if op.type == 'STAR':
                left = left * right
            else:
                if right == 0:
                    raise ExpressionError("Division by zero")
                left = left / right

        return left

    def factor():
        if peek().type == 'MINUS':
            consume()
            return - factor()
        left = primary()
        return left

    def primary():
        token = peek()
        if token.type == 'LPAREN':
            consume()
            result = expression()
            if state['position'] >= len(tokens) or tokens[state['position']].type != 'RPAREN':
                raise ExpressionError("Unbalanced parentheses")
            consume()
            return result
        elif token.type == 'NUMBER':
            return consume().value
        else:
            raise ExpressionError("Unexpected token")

    result = expression()
    if tokens[state['position']].type != 'EOF':
        raise ExpressionError("Trailing characters in expression")

    return result


def lex(expr):
    tokens = []
    i = 0
    while i < len(expr):
        char = expr[i]
        if char.isspace():
            i += 1
        elif char.isdigit() or char == '.':
            num = ""
            while i < len(expr) and (expr[i].isdigit() or expr[i] == '.'):
                num += expr[i]
                i += 1
            tokens.append(Token('NUMBER', float(num)))
        elif char == '+':
            tokens.append(Token('PLUS', '+'))
            i += 1
        elif char == '-':
            tokens.append(Token('MINUS', '-'))
            i += 1
        elif char == '*':
            tokens.append(Token('STAR', '*'))
            i += 1
        elif char == '/':
            tokens.append(Token('SLASH', '/'))
            i += 1
        elif char == '(':
            tokens.append(Token('LPAREN', '('))
            i += 1
        elif char == ')':
            tokens.append(Token('RPAREN', ')'))
            i += 1
        else:
            raise ExpressionError(f"Unexpected character: {char}")
    tokens.append(Token('EOF', 'EOF'))
    return tokens
