Using `namedtuple` is the correct fix. Patching each of those 12 call sites with index access (`token[0]`) is brittle and makes the code harder to read—the parser already expresses itself naturally in terms of `.type` and `.value`, so the lexer should emit that shape. Adding namedtuple in the lexer preserves the existing parser logic while fixing the crash across all rules at once.

```file:calc.py
from collections import namedtuple

Token = namedtuple('Token', ['type', 'value'])

class ExpressionError(Exception):
    pass


def evaluate(expr):
    if not expr.strip():
        raise ExpressionError("Empty expression")

    tokens = lex(expr)

    if not tokens:
        raise ExpressionError("Could not parse")

    position = 0

    def peek():
        return tokens[position]

    def consume():
        token = peek()
        position += 1
        return token

    def expression():
        left = term()
        while position < len(tokens) and peek().type in ('PLUS', 'MINUS'):
            op = consume()
            right = term()
            left = left + right if op.type == 'PLUS' else left - right

        if position < len(tokens) and peek().type not in ('EOF', 'PLUS', 'MINUS'):
            raise ExpressionError("Expected end of expression")

        return left

    def term():
        left = factor()
        while position < len(tokens) and peek().type in ('STAR', 'SLASH'):
            op = consume()
            right = factor()
            if op.type == 'STAR':
                left = left * right
            else:
                if right == 0:
                    raise ExpressionError("Division by zero")
                left = left / right

        if position < len(tokens) and peek().type not in ('EOF', 'PLUS', 'MINUS', 'STAR', 'SLASH'):
            raise ExpressionError("Expected end of term")

        return left

    def factor():
        if peek().type == 'MINUS':
            consume()
            return - factor()
        left = primary()
        if position < len(tokens) and peek().type not in ('EOF', 'PLUS', 'MINUS', 'STAR', 'SLASH'):
            raise ExpressionError("Expected end of factor")
        return left

    def primary():
        token = peek()
        if token.type == 'LPAREN':
            consume()
            result = expression()
            if position >= len(tokens) or tokens[position].type != 'RPAREN':
                raise ExpressionError("Unbalanced parentheses")
            consume()
            return result
        elif token.type == 'NUMBER':
            return consume().value
        else:
            raise ExpressionError("Unexpected token")

    result = expression()
    if position != len(tokens):
        raise ExpressionError("Trailing characters in expression")

    return result


def lex(expr):
    tokens = []
    i = 0
    while i < len(expr):
        char = expr[i]
        if char.isspace():
            i += 1
        elif char.isdigit() or char == '.':
            num = ""
            while i < len(expr) and (expr[i].isdigit() or expr[i] == '.'):
                num += expr[i]
                i += 1
            tokens.append(Token('NUMBER', float(num)))
        elif char == '+':
            tokens.append(Token('PLUS', '+'))
            i += 1
        elif char == '-':
            if tokens and tokens[-1].type == 'NUMBER':
                tokens.append(Token('MINUS', '-'))
                i += 1
            else:
                tokens.append(Token('NEGATIVE', '-'))
                i += 1
        elif char == '*':
            tokens.append(Token('STAR', '*'))
            i += 1
        elif char == '/':
            tokens.append(Token('SLASH', '/'))
            i += 1
        elif char == '(':
            tokens.append(Token('LPAREN', '('))
            i += 1
        elif char == ')':
            tokens.append(Token('RPAREN', ')'))
            i += 1
        else:
            raise ExpressionError(f"Unexpected character: {char}")
    tokens.append(Token('EOF', 'EOF'))
    return tokens
```
